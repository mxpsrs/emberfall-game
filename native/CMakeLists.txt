cmake_minimum_required(VERSION 3.20)
project(veldren_native LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ZLIB REQUIRED)

add_library(veldren_core STATIC src/core.cpp src/desktop.cpp src/scene.cpp src/assets.cpp src/asset_render.cpp src/asset_material.cpp src/asset_import.cpp src/asset_texture.cpp)
target_include_directories(veldren_core PUBLIC include)
target_include_directories(veldren_core SYSTEM PRIVATE third_party)

add_executable(veldren-desktop src/desktop_main.cpp src/desktop_renderer.cpp)
target_link_libraries(veldren-desktop PRIVATE veldren_core ZLIB::ZLIB ${CMAKE_DL_LIBS})

add_executable(veldren-desktop-test tests/desktop_test.cpp)
target_link_libraries(veldren-desktop-test PRIVATE veldren_core)
add_executable(veldren-scene-test tests/scene_test.cpp)
target_link_libraries(veldren-scene-test PRIVATE veldren_core)
add_executable(veldren-desktop-scene-test tests/desktop_scene_test.cpp)
target_link_libraries(veldren-desktop-scene-test PRIVATE veldren_core)
add_executable(veldren-scene-migrate tools/migrate_scene.cpp)
target_link_libraries(veldren-scene-migrate PRIVATE veldren_core)

if(MSVC)
  target_compile_options(veldren_core PRIVATE /W4 /WX)
  target_compile_options(veldren-desktop PRIVATE /W4 /WX)
  target_compile_options(veldren-desktop-test PRIVATE /W4 /WX)
  target_compile_options(veldren-scene-test PRIVATE /W4 /WX)
  target_compile_options(veldren-desktop-scene-test PRIVATE /W4 /WX)
  target_compile_options(veldren-scene-migrate PRIVATE /W4 /WX)
else()
  target_compile_options(veldren_core PRIVATE -Wall -Wextra -Werror)
  target_compile_options(veldren-desktop PRIVATE -Wall -Wextra -Werror)
  target_compile_options(veldren-desktop-test PRIVATE -Wall -Wextra -Werror)
  target_compile_options(veldren-scene-test PRIVATE -Wall -Wextra -Werror)
  target_compile_options(veldren-desktop-scene-test PRIVATE -Wall -Wextra -Werror)
  target_compile_options(veldren-scene-migrate PRIVATE -Wall -Wextra -Werror)
endif()

enable_testing()
add_test(NAME desktop-core COMMAND veldren-desktop-test)
set_tests_properties(desktop-core PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME scene-core COMMAND veldren-scene-test)
set_tests_properties(scene-core PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME desktop-scene COMMAND veldren-desktop-scene-test)
set_tests_properties(desktop-scene PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME desktop-headless COMMAND veldren-desktop --headless --frames=240 --assets=..)
set_tests_properties(desktop-headless PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})

add_executable(veldren-asset-test tests/assets_test.cpp)
target_link_libraries(veldren-asset-test PRIVATE veldren_core)
add_test(NAME asset-registry COMMAND veldren-asset-test ${CMAKE_CURRENT_SOURCE_DIR}/../dist/assets/asset-registry.json)

add_executable(veldren-asset-import tools/import_asset.cpp)
target_link_libraries(veldren-asset-import PRIVATE veldren_core)

add_executable(veldren-texture-test tests/texture_test.cpp)
target_link_libraries(veldren-texture-test PRIVATE veldren_core)
add_test(NAME asset-textures COMMAND veldren-texture-test ${CMAKE_CURRENT_SOURCE_DIR}/../dist/assets/canonical/images)

add_executable(veldren-texture-variants tools/texture_variants.cpp)
target_link_libraries(veldren-texture-variants PRIVATE veldren_core ZLIB::ZLIB)

add_executable(veldren-material-test tests/material_test.cpp)
target_link_libraries(veldren-material-test PRIVATE veldren_core)
add_test(NAME asset-materials COMMAND veldren-material-test ${CMAKE_CURRENT_SOURCE_DIR}/../dist/assets/asset-registry.json)

add_executable(veldren-render-asset-test tests/render_asset_test.cpp)
target_link_libraries(veldren-render-asset-test PRIVATE veldren_core)
add_test(NAME asset-render COMMAND veldren-render-asset-test ${CMAKE_CURRENT_SOURCE_DIR}/../dist/assets)
